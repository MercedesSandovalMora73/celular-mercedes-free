import os, sys from PyQt6.QtCore import QUrl from PyQt6.QtWidgets import QApplication, QMainWindow from PyQt6.QtWebEngineWidgets import QWebEngineView class NavegadorIO(QMainWindow): def __init__(self): super().__init__() self.setWindowTitle("IO INTELIGENCIA OMICA - Navegador") self.resize(450,800) view=QWebEngineView() view.load(QUrl.fromLocalFile(os.path.abspath("index.html"))) self.setCentralWidget(view) if __name__=="__main__": app=QApplication(sys.argv) v=NavegadorIO() v.show() sys.exit(app.exec())
